void ordenarSociosId() {


  for (int i = 0; i < 5; i++) {
    for (int j = i; j < 5; j++) {


      if (socios[i].id < socios[j].id) {


        Socio t = socios[i];
        socios[i] = socios[j];
        socios[j] = t;
      }
    }
  }
}


void ordenarSociosParticipacao() {


  for (int i = 0; i < 5; i++) {
    for (int j = i; j < 5; j++) {


      if (socios[i].participacaoSocietaria < socios[j].participacaoSocietaria) {


        Socio t = socios[i];
        socios[i] = socios[j];
        socios[j] = t;
      }
    }
  }
}


void exibirDadosSocios() {


  int isValido = 1;


  while (isValido) {
    system("clear");


    printf("Lista de Sócios\nId |   Nome   | Sexo |      Cargo      | "
           "Participação acionária\n");


    for (int i = 0; i < 5; i++) {


      printf("%2d | %-8s | %-4s | %-15s | %.2f% \n", socios[i].id,
             socios[i].nome, socios[i].sexo, socios[i].cargo,
             socios[i].participacaoSocietaria);
    }


    int escolha;


    printf("\n1.Ordenar por Id decrescente\n2.Ordenar por Participação "
           "decrescente\n3.Voltar\n");


    scanf("%d", &escolha);


    switch (escolha) {


    case 1:
      ordenarSociosId();
      break;
    case 2:
      ordenarSociosParticipacao();
      break;
    case 3:
      isValido = 0;
      break;
    }
  }
}
